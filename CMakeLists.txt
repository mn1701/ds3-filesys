cmake_minimum_required(VERSION 3.16)
project(ds3 C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
add_executable(ds3 src/main.c src/codec.c src/storage.c src/inodes.c src/files.c)
target_include_directories(ds3 PRIVATE include)
if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(ds3 PRIVATE -Wall -Wextra -Wpedantic)
endif()
